// Copy to internal/api/url_evidence_tmp_test.go, run only this test, then remove it.
func TestURLSyncEvidence(t *testing.T) {
	path := "../../docs/protocol-url-export/evidence/"
	raw, err := os.ReadFile(path + "after-urls.json")
	if err != nil {
		t.Fatal(err)
	}
	var samples []struct {
		Name    string                     `json:"name"`
		Mihomo  string                     `json:"mihomo_yaml"`
		Profile serverconfig.ClientProfile `json:"profile"`
	}
	if err := json.Unmarshal(raw, &samples); err != nil {
		t.Fatal(err)
	}
	output := map[string]string{}
	for _, sample := range samples {
		if !sample.Profile.SubscriptionCompatible {
			continue
		}
		sample.Profile.Mihomo = sample.Mihomo
		value, err := renameSubStoreNode(subStoreClientValue(sample.Profile), "同步 🚀 + %2F @:/?#&=")
		if err != nil {
			t.Fatal(sample.Name, err)
		}
		output[sample.Name] = value
	}
	raw, err = json.MarshalIndent(output, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path+"sync-values.json", raw, 0600); err != nil {
		t.Fatal(err)
	}
}
